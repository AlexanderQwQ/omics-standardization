# =============================================================================
# omics_standardization — Environment Setup Guide
# =============================================================================
#
# This file documents the complete dependency list and setup instructions.
# The PRIMARY way to create the environment is via conda:
#
#     conda env create -f environment.yml
#     conda activate omics-std
#     pip install -e ".[test,dev]"
#
# The environment.yml includes ALL dependencies: Python, PyTorch, R,
# Bioconductor, kallisto, and all file-format parsers.
#
# If you prefer pip + venv, see the manual setup steps below.
# =============================================================================


# =============================================================================
# Alternative: pip + venv (Python packages only, no R / kallisto)
# =============================================================================
#
# 1. Create venv:
#    python -m venv .venv
#    source .venv/bin/activate        # Linux/macOS
#    .venv\Scripts\activate           # Windows
#
# 2. Install Python deps:
#    pip install -r requirements.txt
#
# 3. Install project in editable mode:
#    pip install -e ".[test,dev]"
#
# NOTE: The following system-level prerequisites are NOT managed by pip
# and must be installed separately for full functionality.
# =============================================================================


# #############################################################################
# System-level prerequisites (manual install — NOT managed by pip)
# #############################################################################
#
# 1. R + Bioconductor (needed by rpy2 for TMM / DESeq2 / VSN / Scran)
#    - Windows: https://cran.r-project.org/bin/windows/base/
#    - macOS:   brew install r
#    - Linux:   sudo apt install r-base
#    Then in R:
#      if (!require("BiocManager", quietly = TRUE)) install.packages("BiocManager")
#      BiocManager::install(c("edgeR", "DESeq2", "limma", "scran"))
#
# 2. kallisto (needed for FASTQ paired-end quantification, must be on PATH)
#    - conda:  conda install -c bioconda kallisto
#    - manual: https://pachterlab.github.io/kallisto/download
#
# 3. DM8 database client (Windows only, optional — SQLite is the default)
#    Install DM8 client before uncommenting dmPython below.
# #############################################################################
